if(APPLE)
  set(OPENSSL_USE_STATIC_LIBS TRUE)
endif()
find_package(OpenSSL ${REQUIRED_OPENSSL_VERSION} REQUIRED COMPONENTS SSL Crypto)
add_library(streaming STATIC
  Protocol.cpp Protocol.h
  SessionBroker.cpp SessionBroker.h
  SecureChannel.cpp SecureChannel.h
  PrivateIpc.cpp PrivateIpc.h
  Service.cpp Service.h
  Control.cpp Control.h
)
target_link_libraries(streaming PUBLIC Qt6::Gui Qt6::Network common OpenSSL::SSL OpenSSL::Crypto PRIVATE base)
if(WIN32)
  target_sources(streaming PRIVATE WindowsControl.cpp WindowsPrivateIpc.cpp WindowsPrivateIpc.h)
  target_link_libraries(streaming PRIVATE advapi32 version dwmapi user32 wtsapi32)
elseif(APPLE)
  target_sources(streaming PRIVATE MacControl.mm)
  set_source_files_properties(MacControl.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
  target_link_libraries(streaming PRIVATE "-framework AppKit" "-framework ApplicationServices" "-framework Carbon")
elseif(BUILD_X11_SUPPORT)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(CONTROL_XCB REQUIRED IMPORTED_TARGET xcb xcb-randr xcb-xinput xcb-xtest xcb-keysyms)
  target_sources(streaming PRIVATE X11Control.cpp)
  target_link_libraries(streaming PRIVATE PkgConfig::CONTROL_XCB Qt6::DBus)
else()
  target_sources(streaming PRIVATE UnavailableControl.cpp)
endif()

option(BUILD_STREAMING "Build GStreamer media capture adapters" OFF)
add_library(streaming-capture STATIC Capture.cpp Capture.h)
target_link_libraries(streaming-capture PUBLIC Qt6::Gui)
target_include_directories(streaming-capture PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
if(BUILD_STREAMING)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GSTREAMER REQUIRED IMPORTED_TARGET
    gstreamer-1.0>=1.28.7 gstreamer-app-1.0>=1.28.7 gstreamer-video-1.0>=1.28.7
    gstreamer-audio-1.0>=1.28.7 gstreamer-webrtc-1.0>=1.28.7 gstreamer-sdp-1.0>=1.28.7 gstreamer-rtp-1.0>=1.28.7 nice)
  target_sources(streaming-capture PRIVATE GstCapturePipeline.cpp GstCapturePipeline.h GstRuntime.cpp GstRuntime.h)
  target_compile_definitions(streaming-capture PRIVATE
    DESKFLOW_STREAMING_BUILD_DIRECTORY="${CMAKE_BINARY_DIR}")
  target_sources(streaming-capture PRIVATE FileSource.cpp FileSource.h)
  target_sources(streaming-capture PRIVATE Audio.cpp Audio.h MediaTransport.cpp MediaTransport.h)
  target_link_libraries(streaming-capture PUBLIC streaming)
  target_compile_definitions(streaming-capture PRIVATE GST_USE_UNSTABLE_API)
  target_sources(streaming-capture PRIVATE NativeAudio.h)
  target_link_libraries(streaming-capture PUBLIC PkgConfig::GSTREAMER)
  target_compile_definitions(streaming-capture PUBLIC DESKFLOW_CAPTURE_GSTREAMER)
endif()
if(WIN32)
  target_sources(streaming-capture PRIVATE WindowsCapture.cpp WindowsWgcSource.cpp WindowsWgcSource.h WindowsSessionEnvironment.cpp)
  target_link_libraries(streaming-capture PRIVATE dwmapi wtsapi32 windowsapp shcore d3d11 dxgi ole32 powrprof)
  if(BUILD_TESTS)
    find_package(Qt6 REQUIRED COMPONENTS Test Widgets)
    add_executable(StreamingControlProbe "${CMAKE_SOURCE_DIR}/tools/ControlProbe.cpp")
    target_link_libraries(StreamingControlProbe PRIVATE streaming Qt6::Widgets Qt6::Test dwmapi)
    add_executable(StreamingCaptureProbe "${CMAKE_SOURCE_DIR}/tools/CaptureProbe.cpp")
    target_link_libraries(StreamingCaptureProbe PRIVATE streaming-capture)
    if(BUILD_STREAMING)
      find_package(Qt6 REQUIRED COMPONENTS Test)
      add_executable(StreamingPackageProbe "${CMAKE_SOURCE_DIR}/tools/PackageProbe.cpp")
      target_link_libraries(StreamingPackageProbe PRIVATE streaming-capture)
      add_executable(StreamingAudioProbe "${CMAKE_SOURCE_DIR}/tools/AudioProbe.cpp")
      target_link_libraries(StreamingAudioProbe PRIVATE streaming-capture Qt6::Test)
    endif()
  endif()
elseif(APPLE)
  target_sources(streaming-capture PRIVATE MacCapture.mm MacSessionEnvironment.mm)
  set_source_files_properties(MacCapture.mm MacSessionEnvironment.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
  target_link_libraries(streaming-capture PRIVATE "-framework ScreenCaptureKit" "-framework CoreMedia"
    "-framework CoreVideo" "-framework AppKit")
  if(BUILD_STREAMING)
    target_sources(streaming-capture PRIVATE MacAudio.mm)
    set_source_files_properties(MacAudio.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
    target_link_libraries(streaming-capture PRIVATE "-framework CoreAudio")
  endif()
else()
  target_sources(streaming-capture PRIVATE LinuxCapture.cpp LinuxSessionEnvironment.cpp)
  target_link_libraries(streaming-capture PRIVATE Qt6::DBus)
  if(BUILD_STREAMING)
    pkg_check_modules(CAPTURE_PIPEWIRE REQUIRED IMPORTED_TARGET libpipewire-0.3>=1.0)
    target_sources(streaming-capture PRIVATE LinuxAudio.cpp)
    target_link_libraries(streaming-capture PRIVATE PkgConfig::CAPTURE_PIPEWIRE)
    pkg_check_modules(CAPTURE_PORTAL REQUIRED IMPORTED_TARGET libportal>=0.9.1)
    pkg_check_modules(CAPTURE_XCB REQUIRED IMPORTED_TARGET xcb xcb-composite xcb-randr xcb-xfixes)
    target_sources(streaming-capture PRIVATE PortalCapture.cpp X11Capture.cpp)
    target_link_libraries(streaming-capture PRIVATE PkgConfig::CAPTURE_PORTAL PkgConfig::CAPTURE_XCB Qt6::DBus)
  endif()
endif()
